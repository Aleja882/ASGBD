CREATE PROCEDURE obtenerEmpleadoMarketing()
BEGIN 
    DECLARE nombreMarketing varchar(20)
    SET nombreMarketing = 'MKT';
    SELECT*FROM empleados WHERE dpto= nombreMarketing;
END;
